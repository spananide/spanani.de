# /flags Implementation Plan

> **For agentic workers:** Execute task by task. Per the owner's rule each task is one Gemini run (gemini skill); Claude reviews the diff and runs the check.

**Goal:** A standalone flag learning app at `spanani.de/flags/` covering 243 flags with 5 levels and 7 modes.

**Architecture:** Static ES modules, no build. Pure logic in `engine.js` (node-testable), generated data in `data.js`, one module per screen, shared `store.js`/`i18n.js`. A one-off generator script builds `data.js` and copies the SVGs from the `flag-icons` npm package.

**Spec:** `docs/superpowers/specs/2026-10-30-flags-design.md`

### Task 2: Engine (TDD)

**Files:** Create `flags/engine.js`, `flags/engine.test.mjs`. No DOM, ES module, imports only `./data.js`.

**Interfaces (exact):**
```js
export const LEVELS = {
  starter:    { mode:'choice', n:3, tiers:[1],           rounds:10, hint:true },
  easy:       { mode:'choice', n:4, tiers:[1,2],         rounds:10 },
  medium:     { mode:'choice', n:4, tiers:[1,2,3,4],     rounds:15, lookalike:true },
  hard:       { mode:'type',        tiers:[1,2,3,4,5],   rounds:20 },
  impossible: { mode:'type',        tiers:[4,5],         rounds:Infinity, timer:10, lives:3, crop:0.5, withLike:true },
};
export function norm(s)                        // string -> string
export function matches(answer, flag, flags=FLAGS) // boolean
export function poolFor(level, continent=null, flags=FLAGS, like=LIKE) // flag[]
export function choicesFor(flag, level, pool, rand=Math.random, flags=FLAGS, like=LIKE) // flag[], length LEVELS[level].n, contains flag, shuffled
export function rng(seed)                      // mulberry32: seed int -> () => float in [0,1)
export function dailyPick(dateStr, n=10, flags=FLAGS) // 'YYYY-MM-DD' -> n distinct flags (tiers 1-4), same input = same output
export function nextBox(box, correct)          // int 0-5 -> int 1-5
export function pickLearn(boxes, flags, n, rand=Math.random) // boxes: {code:box}; -> n distinct flags, weight 6-box (unseen = box 0 = weight 6)
export function neighbours(code, like=LIKE)    // codes sharing a LIKE group with code (no duplicates, excludes itself)
```
Semantics:
- `norm`: lowercase, `ß`->`ss`, NFD + strip combining marks (so `ä ö ü` -> `a o u`), `&`->` and `, remove text in parentheses, map the word `st`/`st.` to `saint`, replace every character outside `a-z0-9` with a space, collapse spaces, trim (so `Guinea-Bissau` -> `guinea bissau`). Examples: `norm("Côte d’Ivoire")==="cote d ivoire"`, `norm("St. Lucia")==="saint lucia"`, `norm("Trinidad & Tobago")==="trinidad and tobago"`, `norm("Türkei")==="turkei"`, `norm("  FRANCE ")==="france"`.
- `matches`: `a=norm(answer)`; names = norm of `en`, `de`, and each `al`. If `a` empty -> false. Exact equal to any name of `flag` -> true. Otherwise if `a` exactly equals any name of ANOTHER flag in `flags` -> false. Otherwise Levenshtein distance to each name of `flag`: allowed 0 if name length < 5, 1 if length 5-8, 2 if length >= 9; true if distance <= allowed.
- `poolFor`: flags whose `t` in `LEVELS[level].tiers`, plus (if `withLike`) every flag whose code appears in any LIKE group; filtered to `k===continent` when continent given.
- `choicesFor`: n-1 distinct distractors, never the flag itself. Only when `LEVELS[level].lookalike` is true, first take distractors from `neighbours(flag.c)` that are in `pool` (up to n-1, random order). Fill the rest from `pool` flags of the same continent, then from the rest of `pool`; if `pool` is too small fall back to `flags`. Shuffle the result (Fisher-Yates with `rand`).
- `dailyPick`: seed = sum over chars of `dateStr` hashed (e.g. `h=h*31+code>>>0`), `rng(seed)`, shuffle copy of tier 1-4 flags sorted by `c` (deterministic), take n.
- `nextBox(box, correct)`: correct -> `Math.min(5, Math.max(box,1)+1)`; wrong -> 1.
- `pickLearn`: weighted sampling without replacement, weight `6 - (boxes[c]||0)`.

**Test file** (`node --test flags/engine.test.mjs`) must contain at least these tests, written first and run red before the engine exists:
```js
import test from 'node:test'; import assert from 'node:assert/strict';
import { FLAGS, LIKE } from './data.js';
import { LEVELS, norm, matches, poolFor, choicesFor, rng, dailyPick, nextBox, pickLearn, neighbours } from './engine.js';
const F = c => FLAGS.find(f => f.c === c);

test('norm', () => {
  assert.equal(norm("Côte d’Ivoire"), 'cote d ivoire');
  assert.equal(norm('St. Lucia'), 'saint lucia');
  assert.equal(norm('Trinidad & Tobago'), 'trinidad and tobago');
  assert.equal(norm('Türkei'), 'turkei');
  assert.equal(norm('  FRANCE '), 'france');
  assert.equal(norm('Myanmar (Burma)'), 'myanmar');
});
test('matches: names, german, alias, typo', () => {
  assert.ok(matches('Germany', F('de')));
  assert.ok(matches('deutschland', F('de')));
  assert.ok(matches('usa', F('us')));
  assert.ok(matches('Frankreich', F('fr')));
  assert.ok(matches('Netherlnds', F('nl')));      // 1 typo, length > 5
  assert.ok(!matches('', F('fr')));
  assert.ok(!matches('xyz', F('fr')));
});
test('matches: another flag\'s exact name is never a typo', () => {
  assert.ok(!matches('Niger', F('ng')));          // Nigeria
  assert.ok(!matches('Nigeria', F('ne')));        // Niger
  assert.ok(!matches('Sudan', F('ss')));          // South Sudan
  assert.ok(!matches('Dominica', F('do')));
  assert.ok(!matches('Guinea', F('gw')));
});
test('matches: short names get no typo slack', () => {
  assert.ok(!matches('Chas', F('td')));           // "chad" has 4 letters: no slack
  assert.ok(!matches('Iran', F('iq')));           // Iraq
});
test('poolFor', () => {
  assert.equal(poolFor('starter').length, 30);
  assert.ok(poolFor('easy').every(f => [1,2].includes(f.t)));
  assert.ok(poolFor('hard').length === FLAGS.length);
  assert.ok(poolFor('medium', 'Africa').every(f => f.k === 'Africa'));
  const imp = poolFor('impossible');
  assert.ok(imp.some(f => f.c === 'it'));          // tier 1 but in a look-alike group
  assert.ok(imp.some(f => f.t === 5));
});
test('choicesFor', () => {
  const r = rng(7);
  for (const lvl of ['starter','easy','medium']) {
    const pool = poolFor(lvl);
    const f = pool[3];
    const ch = choicesFor(f, lvl, pool, r);
    assert.equal(ch.length, LEVELS[lvl].n);
    assert.equal(new Set(ch.map(x => x.c)).size, ch.length);
    assert.ok(ch.includes(f));
  }
  const ro = F('ro'); const ch = choicesFor(ro, 'medium', poolFor('medium'), rng(1));
  assert.ok(ch.some(x => neighbours('ro').includes(x.c)));   // look-alike distractor present
});
test('neighbours', () => {
  assert.ok(neighbours('ro').includes('td'));
  assert.ok(!neighbours('ro').includes('ro'));
  assert.deepEqual(neighbours('us').filter((c, i, a) => a.indexOf(c) !== i), []);
});
test('rng is deterministic', () => {
  const a = rng(42), b = rng(42);
  assert.deepEqual([a(), a(), a()], [b(), b(), b()]);
});
test('dailyPick', () => {
  const a = dailyPick('2026-10-30'), b = dailyPick('2026-10-30'), c = dailyPick('2026-10-31');
  assert.deepEqual(a.map(f => f.c), b.map(f => f.c));
  assert.notDeepEqual(a.map(f => f.c), c.map(f => f.c));
  assert.equal(new Set(a.map(f => f.c)).size, 10);
  assert.ok(a.every(f => f.t <= 4));
});
test('nextBox', () => {
  assert.equal(nextBox(0, true), 2);
  assert.equal(nextBox(3, true), 4);
  assert.equal(nextBox(5, true), 5);
  assert.equal(nextBox(4, false), 1);
  assert.equal(nextBox(0, false), 1);
});
test('pickLearn prefers weak flags', () => {
  const boxes = Object.fromEntries(FLAGS.map(f => [f.c, 5])); boxes.fr = 1;
  let hits = 0; const r = rng(3);
  for (let i = 0; i < 200; i++) if (pickLearn(boxes, FLAGS, 5, r).some(f => f.c === 'fr')) hits++;
  assert.ok(hits > 100);
  const p = pickLearn({}, FLAGS, 10, rng(1));
  assert.equal(new Set(p.map(f => f.c)).size, 10);
});
```
